int numOccurence(string p , string s){
    int count = 0;
    
    for(int i=0;i<s.size()-p.size();i++){
        bool has =true;
        for(int j=0;j<p.size();j++){
            if(s[i+j]!=p[j])has=false;
        }
        if(has)count++;
    }
    
    return count;
}

    string s = "abceacfcacb";
    cout<<numOccurence(p,s);

/*
Given an integer array, find the contiguous subarray with the largest sum and return that sum.
*/

// Function to find the maximum subarray sum
int maxSubArray(vector<int>& nums) {

    int maxSum=INT_MIN;
    int sum=0;
    
    for(int i=0;i<nums.size();i++){
        sum+=nums[i];
        if(sum>maxSum)maxSum=sum;
        if(sum<0)sum=0;
    }
    return maxSum; // Placeholder
}

int main() {
    // Test case 1
    vector<int> nums1 = {-2, 1, -3, 4, -1, 2, 1, -5, 4};
    cout << "Test case 1: " << maxSubArray(nums1) << endl; // Expected: 6

    // Test case 2
    vector<int> nums2 = {1};
    cout << "Test case 2: " << maxSubArray(nums2) << endl; // Expected: 1

public class CipherTest {

    // Function template: takes key and cipher string, returns decoded string
    public static String decodeCipher(int key, String cipherText) {
        
        StringBuilder sb = new StringBuilder();
        for(char c : cipherText.toCharArray()){
            c= (char) (c-key);
            sb.append(c);
        }
        
        return sb.toString();
    }

    public static void main(String[] args) {
