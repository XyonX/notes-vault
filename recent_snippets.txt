class Solution {
public:
    double findMaxAverage(vector<int>& nums, int k) {


        int windowSum=0;
        for(int i=0;i<k;i++){
            windowSum+=nums[i];
        }
        double maxAvg = double(windowSum)/k;


        for(int i=1;i<=nums.size()-k;i++){
            windowSum-=nums[i-1];
            windowSum+=nums[i+k-1];
            
            double avg=double(windowSum)/k;
            maxAvg=max(avg,maxAvg);
        }

        return maxAvg;
        
    }
};

int longestSubarraySameElement(const vector<int>& arr){
    
    if(arr.size()==0)return 0;
        int maxLength=1;
    
    int l=0;

    for(int r=1;r<arr.size();r++){
        
        if(arr[r] != arr[r-1]){
            l=r;
        }
        
        maxLength=max(maxLength,r-l+1);
        
    }
    return maxLength;
}

// Function to reverse words in a string (to be implemented)
string reverseWords(string s) {
    
    //remove leading spaces
    int from=0;
    while(from <s.size() && s[from] ==' '){
        from++;
    }
    //remove training spaces
    int to=s.size()-1;
    while(to>=0 && s[to]==' '){
        to--;
    }
    s=s.substr(from,to-from+1);
    
    reverse(s.begin(),s.end());
    s+=" ";
    int starting =0;
    for(int i=1;i<s.size();i++){
        if(s[i]==' '){
            reverse(s.begin()+starting, s.begin()+i);
            starting =i+1;
        }
    }
    
    s.pop_back();
    return s;
    
}
